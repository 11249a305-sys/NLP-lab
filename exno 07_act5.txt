sentence = ["the", "cat", "runs"]

states = ['DT', 'NN', 'VB']

V = [{}]
path = {}

# Initialization
for state in states:
    V[0][state] = start_p[state] * emit_p[state].get(sentence[0], 0)
    path[state] = [state]

# Recursion
for t in range(1, len(sentence)):
    V.append({})
    new_path = {}

    for state in states:
        (prob, prev_state) = max(
            (V[t-1][prev] * trans_p[prev][state] *
             emit_p[state].get(sentence[t], 0), prev)
            for prev in states
        )

        V[t][state] = prob
        new_path[state] = path[prev_state] + [state]

    path = new_path

# Termination
n = len(sentence) - 1
prob, state = max((V[n][state], state) for state in states)

print("Most likely POS tags:", path[state])
