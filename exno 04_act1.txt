# DFA for ab*

import matplotlib.pyplot as plt

plt.figure(figsize=(6, 3))

# States
plt.scatter([1, 3, 5], [1, 1, 1], s=1500)

plt.text(1, 1, "q0", ha="center")
plt.text(3, 1, "q1", ha="center")
plt.text(5, 1, "q2", ha="center")

# Transitions
plt.annotate("a", (2.7, 1.1), ha="center")
plt.annotate("b", (3, 1.3), ha="center")
plt.annotate("b", (4, 1.1), ha="center")
plt.annotate("a", (4, 0.8), ha="center")

plt.title("DFA for ab*")
plt.axis("off")
plt.show()
