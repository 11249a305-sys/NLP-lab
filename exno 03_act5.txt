import re

text = """
Aadhaar: 123456789012
PAN: ABCDE1234F
Vehicle: TN01AB1234
Date: 24-07-2026
Time: 10:30
"""

patterns = {
    "Aadhaar": r'\b\d{12}\b',
    "PAN": r'\b[A-Z]{5}\d{4}[A-Z]\b',
    "Vehicle Number": r'\b[A-Z]{2}\d{2}[A-Z]{1,2}\d{4}\b',
    "Date": r'\b\d{2}-\d{2}-\d{4}\b',
    "Time": r'\b\d{2}:\d{2}\b'
}

for name, pattern in patterns.items():
    result = re.findall(pattern, text)
    print(name, ":", result)
