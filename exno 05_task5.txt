data = [
    ("happy", "happiness", "-ness", "Derivational"),
    ("walk", "walked", "-ed", "Inflectional"),
    ("national", "nationalize", "-ize", "Derivational"),
    ("book", "books", "-s", "Inflectional"),
    ("kind", "unkind", "un-", "Derivational"),
    ("run", "running", "-ing", "Inflectional")
]
print("Task 5 - Inflectional vs. Derivational Classification")
print()
print("{:<12} {:<18} {:<10} {}".format(
    "Base Word", "Derived Word", "Affix", "Type"))
print("-" * 65)
for base, word, affix, typ in data:
    print("{:<12} {:<18} {:<10} {}".format(
        base, word, affix, typ))
