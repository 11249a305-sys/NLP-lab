import spacy
import nltk
from nltk.stem import WordNetLemmatizer
nltk.download('wordnet')
nltk.download('omw-1.4')
# Load spaCy model
nlp = spacy.load("en_core_web_sm")
# Initialize NLTK lemmatizer
lemmatizer = WordNetLemmatizer()
text = """
The children are playing in the gardens.
They went to different schools and studied hard.
The cats were running and eating food.
"""
# spaCy lemmatization
doc = nlp(text)
print("spaCy Lemmatization:")
for token in doc:
    if token.is_alpha:
        print(token.text, "---->", token.lemma_)
print("\nNLTK WordNet Lemmatization:")
for token in doc:
    if token.is_alpha:
        print(
            token.text,
            "---->",
            lemmatizer.lemmatize(token.text.lower())
        )
